from unsloth import FastLanguageModel
import torch
from datasets import load_dataset

# Load your generated Intentra dataset
dataset = load_dataset("json", data_files="intentra_dataset.jsonl")

model, tokenizer = FastLanguageModel.from_pretrained(
    model_name = "lora_model",
    max_seq_length = 2048,
    dtype = None,
    load_in_4bit = True,
)

# Apply LoRA specifically for classification heads
model = FastLanguageModel.get_peft_model(
    model,
    r = 16,
    target_modules = ["q_proj", "k_proj", "v_proj", "o_proj"],
    lora_alpha = 16,
    lora_dropout = 0,
    bias = "none",
)

print("Model ready for intent-driven fine-tuning!")
